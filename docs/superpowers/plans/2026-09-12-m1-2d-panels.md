```cpp
namespace quantviz::viz {
class Scene {                       // Runner + Panel の対を型消去
public:
    virtual ~Scene() = default;
    virtual void start() = 0;       // runner.start()
    virtual void stop() = 0;        // runner.stop()
    virtual bool running() const = 0;
    virtual void draw() = 0;        // 毎フレーム: panel.draw(runner)
};

template <bridge::Model M, class Panel, std::size_t SnapCap = 4096, std::size_t CmdCap = 256>
class RunnerScene final : public Scene {   // 具象シーンの共通実装
public:
    RunnerScene(M model, bridge::RunnerConfig cfg, Panel panel);
    // start/stop/running は runner_ に委譲、draw は panel_.draw(runner_)
private:
    bridge::Runner<M, SnapCap, CmdCap> runner_;
    Panel                              panel_;
};

class SceneRegistry {
public:
    using Factory = std::function<std::unique_ptr<Scene>()>;
    void add(std::string name, Factory make);                 // 登録順を保つ（VIZ-01）
    const std::vector<std::string>& names() const noexcept;
    std::size_t size() const noexcept;
    /// index のシーンを生成して start() する。直前のシーンは stop() してから破棄する（VIZ-02）。

    void select(std::size_t index);
    Scene*      current() noexcept;                            // 未選択なら nullptr
    std::size_t current_index() const noexcept;                // 未選択なら SIZE_MAX
};
}
```

### Snapshot のサイズ方針（M1 の例外）

Greeks / GARCH の Snapshot は固定格子を含むため 128 B を超える（数 KB〜十数 KB）。M2 の triple buffer 導入までは、`RunnerScene` の `SnapCap` を 256 以下に下げ、`RunnerConfig::publish_every` と `steps_per_second` を抑えて帯域を抑える。`static_assert(sizeof(Snapshot) <= 32 * 1024)` を契約テストに入れる。

```cpp
struct BsGreeks { double price, delta, gamma, vega, theta, rho; };
double   bs_price (double S, double K, double T, double r, double sigma, OptionType type) noexcept;

void bs_price_strip(double S, std::span<const double> strikes, double T, double r, double sigma,
                    OptionType type, std::span<double> out) noexcept;
void bs_price_strip_scalar(/* 同じ引数 */) noexcept;  // 参照実装（テストとベンチの比較用）
```
- 極限の扱い: `T <= 0` または `sigma <= 0` では本源的価値 / 決定的な割引ペイオフを返し、Greeks は有限値（Γ, ν は 0）。NaN を返さない。
- SIMD: `#if __has_include(<experimental/simd>)` なら `std::experimental::native_simd<double>`、無ければ `#if defined(__AVX2__)` の intrinsics、どちらも無ければスカラ。Φ のベクトル化は Abramowitz–Stegun 7.1.26 ではなく、**精度 1e-15 を満たすため**スカラ版と同じ式（`erfc`）をレーンごとに呼ぶか、スカラ版と bit 一致するよう `std::experimental::simd` の `erfc` を使う。BS-10 の許容は相対 1e-15 なので、SIMD 版が同じ演算順序でないと落ちる。難しければ許容を「相対 1e-13」に緩める根拠を仕様書側に記録すること（Task 8 に報告）。

```cpp
// optim.hpp
template <std::size_t N> struct OptimResult {
    std::array<double, N> x; double f; std::size_t iters; bool converged;
    std::vector<std::array<double, N>> path;   // 反復ごとの best x（尤度面上の軌跡描画用）
};
struct OptimOptions { std::size_t max_iter = 1000; double tol_f = 1e-10; double tol_x = 1e-8; double initial_step = 0.1; };
template <std::size_t N, class F> OptimResult<N> nelder_mead(F&& f, std::array<double,N> x0, OptimOptions o = {});
template <std::size_t N, class F> OptimResult<N> bfgs(F&& f, std::array<double,N> x0, OptimOptions o = {}); // 数値勾配（中心差分）+ Armijo 直線探索
// 制約変換（往復が恒等）: logit/sigmoid で (0,1)、softplus/逆 で (0,∞)
double to_unit(double x) noexcept; double from_unit(double u) noexcept;        // (0,1) ↔ R
double to_positive(double x) noexcept; double from_positive(double p) noexcept; // (0,∞) ↔ R

// garch.hpp   σ²_t = ω + α r²_{t−1} + β σ²_{t−1}
struct GarchParams { double omega, alpha, beta; };
bool   garch_stationary(GarchParams p) noexcept;  // ω>0, α≥0, β≥0, α+β<1
double garch_unconditional_variance(GarchParams p) noexcept;
/// フィルタ: 各 t の σ²_t を out に書く。初期 σ²_0 = 無条件分散（非定常なら標本分散）。
void   garch_filter(GarchParams p, std::span<const double> r, std::span<double> sigma2_out) noexcept;
double garch_log_likelihood(GarchParams p, std::span<const double> r) noexcept; // −½Σ(log 2π + log σ²_t + r²_t/σ²_t)。非定常なら −inf 相当の大きな負値
/// 無制約パラメータ θ∈R³ ↔ 制約付き (ω,α,β)：ω = softplus, (α,β) は α+β<1 を満たす変換（例: s = sigmoid(θ1) ∈(0,1), α = s·sigmoid(θ2), β = s·(1−sigmoid(θ2))）
GarchParams garch_from_unconstrained(std::array<double,3> th) noexcept;
std::array<double,3> garch_to_unconstrained(GarchParams p) noexcept;
struct GarchFit { GarchParams params; double log_lik; OptimResult<3> optim; };
GarchFit garch_fit(std::span<const double> r, GarchParams init, OptimOptions o = {}, bool use_bfgs = false);
/// 合成 GARCH パスの生成（決定性: core::Rng）
void garch_simulate(GarchParams p, std::uint64_t seed, std::span<double> r_out, std::span<double> sigma2_out) noexcept;
```

```cpp
// math/mat.hpp — 固定サイズ行列（std::array ベース、ヒープなし）
template <std::size_t R, std::size_t C> struct Mat {
    std::array<double, R*C> a{};
    double& operator()(std::size_t i, std::size_t j) noexcept; double operator()(...) const noexcept;
    static Mat identity() noexcept;   // R == C
    Mat<C,R> transpose() const noexcept;
};
template <R,K,C> Mat<R,C> operator*(const Mat<R,K>&, const Mat<K,C>&) noexcept;
+, −, スカラ倍, `inverse()` は 1×1 / 2×2 / 3×3 を閉形式で（それ以上は static_assert）。
template <std::size_t N> bool is_symmetric(const Mat<N,N>&, double tol) noexcept;
template <std::size_t N> bool is_psd(const Mat<N,N>&, double tol) noexcept;  // N ≤ 3: 主小行列式 ≥ −tol

// stats/kalman.hpp — 線形カルマン x_t = F x_{t−1} + w (Q), z_t = H x_t + v (R)
template <std::size_t NX, std::size_t NZ> class Kalman {
    void predict(const Mat<NX,NX>& F, const Mat<NX,NX>& Q) noexcept;
    /// 観測行列 H は時変（回帰なら H = [x_t]）。イノベーションを返す。
    Mat<NZ,1> update(const Mat<NZ,NX>& H, const Mat<NZ,1>& z, const Mat<NZ,NZ>& R) noexcept;  // Joseph 形で P を更新（対称・PSD を保つ）
    const Mat<NX,1>& state() const; const Mat<NX,NX>& cov() const;
    void reset(const Mat<NX,1>& x0, const Mat<NX,NX>& P0) noexcept;
};
```
